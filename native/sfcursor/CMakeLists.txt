cmake_minimum_required(VERSION 3.13)
project(sfcursor CXX)
if(NOT WIN32)
    message(FATAL_ERROR "sfcursor requires Windows")
endif()
set(HASHLINK_DIR "" CACHE PATH "HashLink SDK containing include/hl.h")
add_library(sfcursor SHARED cursor.cpp extra_input.cpp shortcut_input.cpp mouse_input.cpp)
target_include_directories(sfcursor PRIVATE "${HASHLINK_DIR}/include" "${HASHLINK_DIR}/src")
target_link_libraries(sfcursor PRIVATE user32 comctl32)
set_target_properties(sfcursor PROPERTIES PREFIX "" SUFFIX ".hdll")
install(TARGETS sfcursor RUNTIME DESTINATION .)
enable_testing()
add_executable(sfcursor-test tests.cpp)
target_include_directories(sfcursor-test PRIVATE "${HASHLINK_DIR}/include" "${HASHLINK_DIR}/src")
target_link_libraries(sfcursor-test PRIVATE user32)
add_test(NAME confinement-policy COMMAND sfcursor-test)
add_executable(extrabars-policy-test extra_tests.cpp)
add_test(NAME extrabars-routing-policy COMMAND extrabars-policy-test)
add_executable(extrabars-win32-test extra_win32_tests.cpp)
target_include_directories(extrabars-win32-test PRIVATE "${HASHLINK_DIR}/include" "${HASHLINK_DIR}/src")
target_link_libraries(extrabars-win32-test PRIVATE user32 comctl32)
add_test(NAME extrabars-win32-wrapper COMMAND extrabars-win32-test)
add_executable(mouse-win32-test mouse_win32_tests.cpp)
target_include_directories(mouse-win32-test PRIVATE "${HASHLINK_DIR}/include" "${HASHLINK_DIR}/src")
target_link_libraries(mouse-win32-test PRIVATE user32 comctl32)
add_test(NAME mouse-guard-win32 COMMAND mouse-win32-test)
